program {
    var int[] nums = [1, 3, 5, 7, 9];
    var int total = 0;
    var int i = 0;

    while (i < 5) {
        var int r = nums[i] % 2;
        if (r == 0) {
            total = total + nums[i];
        }
        i = i + 1;
    }

    int fibonacci(int n) {
        if (n <= 1) { return n; }
        return fibonacci(n - 1) + fibonacci(n - 2);
    }

    print("Fibonacci(20) = ");
    print(fibonacci(20));
    print("Total pares: ");
    print(total);

    struct Persona {
        int edad;
        int altura;
    }
    var Persona p;
    p.edad = 30;
    print("Edad de persona: ");
    print(p.edad);

}
